RsResult world_preview_shrink(World *world, Direction direction, uint32_t count, WorldShrinkPreview *preview);// 预览收缩边界及将被丢弃的编辑，不修改世界。
RsResult world_shrink_plot(World *world, Direction direction, uint32_t count);// 收缩地块并永久丢弃移除区域；是否确认由调用方决定。
size_t world_volume(const World *world);// 读取已加载区块的方块容量，不是世界水平范围。

void world_visit_sections(const World *world, WorldSectionVisitor visitor, void *context);// 访问当前视窗内已加载且非空的区段。
uint64_t world_revision(const World *world);// 读取世界修订计数。
uint64_t world_topology_revision(const World *world);// 读取拓扑修订计数。
void world_set_change_observer(World *world, WorldChangeObserver observer, void *context);// 设置写入后的只读观察回调；NULL 解除。

bool block_is_piston_base(BlockId id);// 判断是否为活塞底座类型。

bool block_piston_pair(BlockId base, BlockId head);// 判断底座和活塞头类型是否配对。
